IK:=1270101;     
DRAWA:=DATE<=IK;


X_2:=DMA(CLOSE,VOL/SUM(VOL,13));
X_3:=(3*HIGH+2*CLOSE+LOW+OPEN)/7;
X_4:=(X_3-X_2)/X_2*100;
X_5:=DMA(CLOSE,VOL/SUM(VOL,34));
X_6:=(3*HIGH+2*CLOSE+LOW+OPEN)/7;
X_7:=(X_3-X_2)/X_2*100;
STICKLINE(DRAWA AND X_4<=(-15),0,15,2,0),COLORRED;
DRAWTEXT(DRAWA AND X_4<=(-15),8,'主力建仓'),COLORWHITE;

A1111:=0.01*EMA(AMOUNT,13)/EMA(VOL,13);
CYC5:=0.01*EMA(AMOUNT,5)/EMA(VOL,5);
CYC2:=0.01*EMA(AMOUNT,2)/EMA(VOL,2);
降龙:=(CLOSE-A1111)/A1111*1000-95;
伏虎:=(CLOSE-A1111)/A1111*1000+150;
A1112:=FINANCE(1)/REF(FINANCE(1),1)>1.02;
A1113:=HHV(A1112,20)=0;
A1114:=H>L;
A1115:=A1114 OR REF(A1114,1);
A1117:=降龙>300  AND A1115;
A8:=IF(A1115,降龙,0);
A9:=(C-REF(C,1))/REF(C,1)*100;
A10:=(HHV(C,4)-LLV(C,10))/LLV(C,10)*100;
A11:=(HHV(C,4)-LLV(C,13))/LLV(C,13)*100;
A12:=((CLOSE-OPEN)/OPEN)*100;
A13:=((H-MAX(OPEN,C))/OPEN)*100;
A14:=(REF(A9,1)>9.9 OR REF(A9,2)>9.9) AND (REF(A9,3)>9.9 OR REF(A9,4)>9.9);
A15:=A13>3 AND A12>0;
A16:=(REF(C,1)-LLV(L,7))/LLV(L,7)*100;
B1:= (A12<0 AND A13>4);
B2:=NOT(REF(A15,1));
B3:=A16>60;
A17:=A11>75 AND ((B1 AND B2 AND B3 ) OR A9<-9.9) ;
A18:=HHV(A8,5) > 300 AND 降龙< HHV(A8,5)*0.40;
B4:=A16>33 AND REF(降龙,1)<0;
A19:=降龙<150   AND  C < CYC5 AND (A16>60 OR B4 ) ;
A20:=(REF(降龙,1)>0 AND REF(降龙,2)>0) OR (REF(降龙,2)>0 AND REF(降龙,3)>0);
A21:=降龙<-15 AND  C < CYC2 AND A16<60 AND A20;
A23:=A18 OR A19 OR A21 OR A17;

A24:=(C-REF(C,20))/REF(C,20)*100;
A25:=A24>15;

STICKLINE(DRAWA AND A23 AND A25,0,15,2,0),COLORFFFF00;
DRAWTEXT(DRAWA AND A23 AND A25,6,'主力出货'),COLORWHITE;


GSPT1:=(CLOSE-LLV(LOW,36))/(HHV(HIGH,36)-LLV(LOW,36))*100;
GSPT2:=SMA(GSPT1,3,1);
GSPT3:=SMA(GSPT2,3,1);
GSPT4:=SMA(GSPT3,3,1);
GSPT5:=CROSS(GSPT3,GSPT4) AND GSPT3<20;
GSPT6:=CROSS(GSPT4,GSPT3) AND GSPT3>80;
GSPT7:=REF(LOW,1);
GSPT8:=SMA(ABS(LOW-GSPT7),13,1)/SMA(MAX(LOW-GSPT7,0),13,1)*4;
GSPT9:=100-3*SMA((CLOSE-LLV(LOW,75))/(HHV(HIGH,75)-LLV(LOW,75))*100,20,1)+2*SMA(SMA((CLOSE-LLV(LOW,75))/(HHV(HIGH,75)-LLV(LOW,75))*100,20,1),15,1);
GSPT16:=100-3*SMA((OPEN-LLV(LOW,75))/(HHV(HIGH,75)-LLV(LOW,75))*100,20,1)+2*SMA(SMA((OPEN-LLV(LOW,75))/(HHV(HIGH,75)-LLV(LOW,75))*100,20,1),15,1);
GSPT18:=GSPT9<REF(GSPT16,1) AND VOL>REF(VOL,1) AND CLOSE>REF(CLOSE,1);

STICKLINE(DRAWA AND GSPT18 AND COUNT(GSPT18,30)=1,0,15,2,0),COLORFF00FF;
DRAWTEXT(DRAWA AND GSPT18 AND COUNT(GSPT18,30)=1,10,'主力进场'),COLORFF00FF;

妖股:= XG AND NOT(REF(ZT, 1)) AND REF(ZT, 2) AND REF(ZT, 3);
