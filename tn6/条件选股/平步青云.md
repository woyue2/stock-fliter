{平步青云选股公式}
{参数设置}
N:=5;  {定义震荡周期}

{条件1：今日成交量大于昨日成交量3倍以上}
倍量 := V / REF(V, 1) >= 3;

{条件2：量能对应的K线必须是非涨停的阳线}
{非涨停：今日收盘价小于涨停价}
最高涨幅 := H < REF(C, 1) * 1.095 AND H > REF(C, 1) * 1.01;  
阳线 := C > O;  {阳线}
条件2 := 最高涨幅  AND 阳线;

{条件3：K线处于震荡突破最近N天高点}
HIGH_N := HHV(H, N);  {最近N日的最高价}
条件3 := C >= HIGH_N AND REF(C, 1) < REF(HIGH_N, 1);  {今日收盘突破N日高点，且昨日未突破}

{综合选股条件}
平步青云 : 倍量 AND 条件2 AND 条件3;
