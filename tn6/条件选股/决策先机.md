{选股器使用时间限制}

IK:=1270101;     
DRAWA:=DATE<=IK;



{ 1.判断昨日涨停且不超过3连板 }
昨日收盘价 := REF(CLOSE, 1);
昨日开盘价 := REF(OPEN, 1);
昨日最高价 := REF(HIGH, 1);
昨日最低价 := REF(LOW, 1);
昨日涨幅 := (昨日收盘价 / REF(CLOSE, 2) - 1) * 100;
前日收盘价 := REF(CLOSE, 2);
前日涨幅 := (前日收盘价 / REF(CLOSE, 3) - 1) * 100;
大前日收盘价 := REF(CLOSE, 3);
大前日涨幅 := (大前日收盘价 / REF(CLOSE, 4) - 1) * 100;

创业板科创板 := CODELIKE('300') OR CODELIKE('688');
ST板块 := NAMELIKE('ST') OR NAMELIKE('*ST');

{ 根据不同板块设置涨停阈值 }
涨停阈值 := IF(创业板科创板, 19.5, 9.5);
昨日涨停 := 昨日涨幅 >= 涨停阈值;
前日涨停 := IF(创业板科创板, 前日涨幅 >= 19.5, 前日涨幅 >= 9.5);
大前日涨停 := IF(创业板科创板, 大前日涨幅 >= 19.5, 大前日涨幅 >= 9.5);

{ 判断昨日非一字板涨停 }
昨日涨停价 := IF(创业板科创板, ROUND(REF(CLOSE, 2) * 1.2 * 100)/100,
ROUND(REF(CLOSE, 2) * 1.1 * 100)/100);
昨日非一字板 := 昨日最低价 < 昨日涨停价 * 0.998;
非一字板 := 昨日涨停 AND 昨日非一字板;

{ 不超过3连板条件 }
不超过3连板 := 非一字板 AND NOT(大前日涨停);



{ 3.所有均线多头向上 }
MA5 := MA(CLOSE, 5);
A1:= MA(CLOSE, 10);
A2:= MA(CLOSE, 20);
A3:= MA(CLOSE, 60);
A4:= MA(CLOSE, 120);
A5:= MA(CLOSE, 250);
MA5F:=(MA(C,5)>A1)*10;
MA10F:=(A1>A2)*20;
MA20F:=(A2>A3)*20;
MA60F:=(A3>A4)*20;
MA120F:=(A4>=REF(A4,1))*20;
MA250F:=(A5>=REF(A5,1))*10;
多头度:=MA5F + MA10F + MA20F + MA60F + MA120F + MA250F;

均线条件 := 多头度>=40;

{ 4.使用开盘价固定值，避免盘中变化 }
固定开盘价 := OPEN;
开盘涨幅 := (固定开盘价 / REF(CLOSE, 1) - 1) * 100;
高开条件 := 开盘涨幅 >= 3 AND 开盘涨幅 <= 6;

{ 5.使用固定集合竞价数据 }
固定竞价最高价 := HIGH;
集合竞价涨停价:=IF(创业板科创板, ROUND(REF(CLOSE, 1) * 1.2 * 100)/100,ROUND(REF(CLOSE, 1) * 1.1 * 100)/100);
竞价触及涨停:=固定竞价最高价 >= 集合竞价涨停价 * 0.998;


{ 排除ST股票 }
非ST股 := NOT(ST板块);

{ 综合选股条件 }
选股条件 := 不超过3连板  {AND 均线条件} AND 高开条件 AND 竞价触及涨停 AND 非ST股;

DRAWA AND 选股条件;
