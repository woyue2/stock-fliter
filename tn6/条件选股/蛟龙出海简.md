{选股器使用时间限制}

IK:=1270101;     
DRAWA:=DATE<=IK;



资金操盘线:=MA(CLOSE,14);
VOLINCREASE:=VOL>REF(VOL,1);
PRICEINCREASE:=(CLOSE-REF(CLOSE,1))/REF(CLOSE,1)*100>5;
HIGHVOL:=VOL/REF(VOL,1)>1.5;
BUYSIGNAL:=CROSS(CLOSE,资金操盘线) AND VOLINCREASE AND PRICEINCREASE AND HIGHVOL;  

WCOND := WINNER(C)*100 > 90;
SELECT: DRAWA AND BUYSIGNAL AND WCOND;
