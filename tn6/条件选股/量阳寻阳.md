{ 异动收阳试盘选股 - 极致精选(不含双创) }

{ --- 基础数据准备 --- }
实体 := ABS(C - O);
上影线 := H - MAX(O, C);
下影线 := MIN(O, C) - L;
均量20 := MA(V, 20);

{ --- 核心条件 1：昨日放量必须绝对大于此前 3 日的每一天 --- }
昨日异动 := REF(V, 1) > REF(V, 2) AND REF(V, 1) > REF(V, 3) AND REF(V, 1) > REF(V, 4);

{ --- 核心条件 2：今日收阳线确认 --- }
今日收阳 := C > O;

{ --- 条件 3：空间防御（处于年内相对低位） --- }
年内高点 := HHV(H, 250);
非高位区 := C < 年内高点 * 0.8;

{ --- 条件 4：历史主力痕迹评分（过去45天） --- }
上影试盘 := 上影线 > 1.5 * 实体 AND V > 均量20 * 1.5;
下影试盘 := 下影线 > 1.5 * 实体 AND V > 均量20 * 1.1;
大阳脉冲 := (C / REF(C, 1) - 1) * 100 > 7.0 AND V > 均量20 * 1.3;
试盘次数 := COUNT(上影试盘 OR 下影试盘 OR 大阳脉冲, 45);

前三日均量 := (REF(V, 2) + REF(V, 3) + REF(V, 4)) / 3;
量比系数 := REF(V, 1) / 前三日均量;
量比得分 := IF(量比系数 >= 2.0, 60, IF(量比系数 >= 1.5, 40, IF(量比系数 >= 1.0, 20, 0)));
频率得分 := MIN(试盘次数, 3) * 10;
强度加分 := IF(EXIST(大阳脉冲, 45), 10, IF(EXIST(上影试盘, 45), 5, 0));
信心总分 := 量比得分 + 频率得分 + 强度加分;

{ --- 条件 5：板块过滤（剔除创业板、科创板、ST股） --- }
非双创板 := NOT(CODELIKE('300')) AND NOT(CODELIKE('688'));
非ST股 := NOT(NAMELIKE('ST')) AND NOT(NAMELIKE('*ST'));

{ --- 最终综合判定 --- }
及格分数 := 60;
异动收阳选股 : 昨日异动 AND 今日收阳 AND 非高位区 AND 信心总分 >= 及格分数 AND 非双创板 AND 非ST股;
