{ 昨日涨停战法选股指标 - 固定版本 }

{ 1.判断昨日涨停且不超过3连板 }
昨日收盘价 := REF(CLOSE, 1);
昨日开盘价 := REF(OPEN, 1);
昨日最高价 := REF(HIGH, 1);
昨日最低价 := REF(LOW, 1);
昨日涨幅 := (昨日收盘价 / REF(CLOSE, 2) - 1) * 100;
前日收盘价 := REF(CLOSE, 2);
前日涨幅 := (前日收盘价 / REF(CLOSE, 3) - 1) * 100;
大前日收盘价 := REF(CLOSE, 3);
大前日涨幅 := (大前日收盘价 / REF(CLOSE, 4) - 1) * 100;

创业板科创板 := CODELIKE('300') OR CODELIKE('688');
ST板块 := NAMELIKE('ST') OR NAMELIKE('*ST');

{ 根据不同板块设置涨停阈值 }
涨停阈值 := IF(创业板科创板, 19.5, 9.5);
昨日涨停 := 昨日涨幅 >= 涨停阈值;
前日涨停 := IF(创业板科创板, 前日涨幅 >= 19.5, 前日涨幅 >= 9.5);
大前日涨停 := IF(创业板科创板, 大前日涨幅 >= 19.5, 大前日涨幅 >= 9.5);

{ 判断昨日非一字板涨停 }
昨日涨停价 := IF(创业板科创板, ROUND(REF(CLOSE, 2) * 1.2 * 100)/100, 
                 ROUND(REF(CLOSE, 2) * 1.1 * 100)/100);
昨日非一字板 := 昨日最低价 < 昨日涨停价 * 0.998;
非一字板 := 昨日涨停 AND 昨日非一字板;

{ 不超过3连板条件 }
不超过3连板 := 非一字板 AND NOT(大前日涨停);

{ 2.前一日破60日高点 }
选股日60日高点 := REF(HHV(HIGH, 60), 1);
前日破60日高点 := 昨日收盘价 >= 选股日60日高点;

{ 3.使用开盘价固定值，避免盘中变化 }
固定开盘价 := OPEN;
开盘涨幅 := (固定开盘价 / REF(CLOSE, 1) - 1) * 100;
高开条件 := 开盘涨幅 >= 3 AND 开盘涨幅 <= 6;

{ 4.使用固定集合竞价数据 }
固定竞价最高价 := HIGH;
集合竞价涨停价 := IF(创业板科创板, ROUND(REF(CLOSE, 1) * 1.2 * 100)/100, 
                    ROUND(REF(CLOSE, 1) * 1.1 * 100)/100);
竞价触及涨停 := 固定竞价最高价 >= 集合竞价涨停价 * 0.998;

{ 5.竞昨日比使用固定值 - 简化版本 }
固定竞价金额 := VOL * OPEN / 10000; { 直接使用开盘成交量估算竞价金额 }
昨日成交金额 := REF(VOL, 1) * REF(CLOSE, 1) / 10000;
竞昨日比 := 固定竞价金额 / 昨日成交金额 * 100;
竞昨日比条件 := 竞昨日比 >= 10;

{ 排除ST股票 }
非ST股 := NOT(ST板块);

{ 综合选股条件 }
选股条件 := 不超过3连板 AND 前日破60日高点 AND 高开条件 AND 竞价触及涨停 AND 竞昨日比条件 AND 非ST股;

选股条件;
