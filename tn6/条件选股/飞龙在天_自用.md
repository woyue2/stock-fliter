{ 飞龙在天策略 - 移除时间限制永久版 }

{ 1. 定义涨停状态 }
ZT := C >= ZTPRICE(REF(C, 1), 0.1) AND C=H;

{ 2. 定义今日价格突破昨日高位的逻辑 (现价 > 昨日开盘价 且 现价 > 昨日收盘价) }
BREAK_YESTERDAY := DYNAINFO(7) > REF(O, 1) AND DYNAINFO(7) > REF(C, 1);

{ 3. 输出选股信号：
   今日突破昨日调整区间 + 昨日未涨停(断板) + 前三天均为连板涨停 }
妖股: BREAK_YESTERDAY AND NOT(REF(ZT, 1)) AND REF(ZT, 2) AND REF(ZT, 3) AND REF(ZT, 4);
