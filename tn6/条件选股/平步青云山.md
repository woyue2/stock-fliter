{平步青云选股公式}
{参数设置}
N:=5;  {定义震荡周期}

{条件1：今日成交量大于昨日成交量3倍以上}
倍量 := V / REF(V, 1) >= 3;

{条件2：量能对应的K线必须是非封死涨停的、带上影线的阳线}
阳线 := C > O;  {基础条件：阳线}
非涨停 := C < REF(C, 1) * 1.095;  {收盘价未封死涨停(主板标准)，允许最高价曾经冲顶涨停以测试抛压}
有上影线 := H > C;  {最高价大于收盘价，确保带上影线}
{如需上影线足够长(比如大于实体的一半)，可改为：(H-C) > (C-O)*0.5;}

条件2 := 阳线 AND 非涨停 AND 有上影线;

{条件3：K线处于震荡突破最近N天高点}
{这根线上冲的高点，必须是最近5天内的最高点，确立反转趋势}
条件3 := H = HHV(H, N);  

{综合选股条件}
平步青云 : 倍量 AND 条件2 AND 条件3;
