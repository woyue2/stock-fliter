{ 决策先机 - 趋势与资金全能组合版 }

{ --- 基础数据与涨停定义 --- }
创业板科创板 := CODELIKE('300') OR CODELIKE('688');
ST板块 := NAMELIKE('ST') OR NAMELIKE('*ST');
涨停阈值 := IF(创业板科创板, 19.5, 9.5);
昨日涨幅 := (REF(CLOSE, 1) / REF(CLOSE, 2) - 1) * 100;
昨日涨停 := 昨日涨幅 >= 涨停阈值;
昨日最低 := REF(LOW, 1);
昨日涨停价 := IF(创业板科创板, ROUND(REF(CLOSE, 2) * 1.2 * 100)/100, ROUND(REF(CLOSE, 2) * 1.1 * 100)/100);
非一字板 := 昨日涨停 AND 昨日最低 < 昨日涨停价 * 0.998;
不超过3连板 := 非一字板 AND (REF(C, 3)/REF(C, 4)-1)*100 < 涨停阈值;

{ --- 简版逻辑：均线趋势多头 --- }
MA5:=MA(C,5); A1:=MA(C,10); A2:=MA(C,20); A3:=MA(C,60); A4:=MA(C,120); A5:=MA(C,250);
多头度 := (MA5>A1)*10 + (A1>A2)*20 + (A2>A3)*20 + (A3>A4)*20 + (A4>=REF(A4,1))*20 + (A5>=REF(A5,1))*10;
均线条件 := 多头度 >= 40;

{ --- VIP2逻辑：60日新高 + 竞价爆量 --- }
前日破60高 := REF(C, 1) >= REF(HHV(H, 60), 1);
开盘涨幅 := (OPEN / REF(C, 1) - 1) * 100;
高开条件 := 开盘涨幅 >= 3 AND 开盘涨幅 <= 6;
竞价触及涨停 := HIGH >= IF(创业板科创板, ROUND(REF(CLOSE, 1)*1.2*100)/100, ROUND(REF(CLOSE, 1)*1.1*100)/100) * 0.998;
竞昨日比 := (VOL * OPEN / 10000) / (REF(VOL, 1) * REF(C, 1) / 10000) * 100 >= 10;

{ --- 最终综合判定 --- }
决策先机全能版 : 不超过3连板 AND NOT(ST板块) AND 均线条件 AND 前日破60高 AND 高开条件 AND 竞价触及涨停 AND 竞昨日比;
