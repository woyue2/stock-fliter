{杯身+长柄箱体底 选股}

{--- 1. 寻找山峰（杯沿），放宽时间到 60 天内 ---}
PEAK_H := HHV(H, 60);       {60天内的最高点}
PEAK_DAY := HHVBARS(H, 60); {最高点距离今天的天数}

{--- 2. 定义前一个微笑（杯身） ---}
{取消了严格的时间跨度限制，只看是否有明显的拉升}
S1_LOW := LLV(L, 120);      
S1_DAY := LLVBARS(L, 120);  
{条件：坑底必须在山峰左侧，且从坑底到山峰至少有 14% 的涨幅（包容002480的急涨）}
S1_DONE := S1_DAY > PEAK_DAY AND (PEAK_H - S1_LOW) / S1_LOW > 0.14;

{--- 3. 定义“长柄箱体”与“当前回踩” ---}
{条件A：山峰发生在 5 到 55 天之间，代表正在画柄部（包容002480的长达44天横盘）}
IN_S2 := PEAK_DAY >= 5 AND PEAK_DAY <= 55;

{寻找山峰以来的箱体最低点}
BOX_LOW := LLV(L, PEAK_DAY); 

{条件B：箱体振幅不能过大，控制在 25% 以内，证明是横盘洗盘而不是崩盘}
BOX_SHAPE := (PEAK_H - BOX_LOW) / PEAK_H < 0.25;

{条件C：当前价格必须从山峰回撤了至少 8%，并且正好跌到了箱体的下沿附近（距离最低点不超过 5%）}
S2_DROP := C <= PEAK_H * 0.92;
NEAR_BOTTOM := C <= BOX_LOW * 1.05 AND C >= BOX_LOW * 0.98;

{--- 4. 均线空头/底部特征 ---}
{5日线在20日线下方，或者当前收盘价被20日线压制}
S2_LEFT_BOTTOM := MA(C,5) < MA(C,20) OR C < MA(C,20); 


{ 1. 基础数字拆解 }
PL := ROUND2(L, 2); { 针对当日下影线最低点 }
{ 提取各数字位，不足的位默认为0 }
D0 := MOD(INTPART(PL/100), 10);  { 百位 (三位数股票) }
D1 := MOD(INTPART(PL/10), 10);   { 十位 (两位数股票) }
D2 := MOD(INTPART(PL), 10);      { 个位 }
D3 := MOD(INTPART(PL*10), 10);   { 十分位 }
D4 := MOD(INTPART(PL*100), 10);  { 百分位 }

{ ===== 2. 基础形态判断 (对子/顺子/相同数) ===== }
{ 四位基础形态 (如11.22, 12.13)，要求股价 >= 10 }
BASE_4BIT := PL>=10 AND (
    (D1=D2 AND D2=D3 AND D3=D4) OR { AAAA }
    (D1=D2 AND D3=D4) OR           { AABB }
    (D1=D3 AND D2=D4) OR           { ABAB }
    (D1=D4 AND D2=D3) OR           { ABBA }
    (D1=D3 AND ABS(D2-D4)=1)       { ABAC }
);

{ 三位基础形态及尾数规律 (适用于所有股价) }
BASE_3BIT := 
    (D2=D3 AND D3=D4) OR           { AAA }
    (D2=D4) OR                     { ABA }
    (D3=D4) OR                     { CC  }
    (D4=0) OR                      { 尾数0 }
    ((D2=D3-1 AND D3=D4-1) OR (D2=D3+1 AND D3=D4+1)); { 顺子 }

{ ===== 3. 数学运算组合穷举 ===== }
{ A. 一位数带两位小数运算 (关注 D2, D3, D4) }
MATH_1BIT := 
    { -- 加法穷举 -- }
    (D2+D3=D4 OR D2+D4=D3 OR D3+D4=D2) OR
    { -- 乘法穷举 (排除全0情况) -- }
    ((D2*D3=D4 OR D2*D4=D3 OR D3*D4=D2) AND PL>=1) OR
    { -- 求和定值为任意整十数 -- }
    (MOD(D2+D3+D4, 10)=0 AND (D2+D3+D4)>0);

{ B. 两位数带两位小数运算 (关注 D1, D2, D3, D4) }
MATH_2BIT := PL>=10 AND (
    { -- 四位数任意求和定值 -- }
    (MOD(D1+D2+D3+D4, 10)=0 AND (D1+D2+D3+D4)>0) OR
    { -- 两两相加对称 -- }
    (D1+D2 = D3+D4) OR (D1+D4 = D2+D3) OR (D1+D3 = D2+D4) OR
    { -- 三加一等于第四数 -- }
    (D1+D2+D3=D4 OR D1+D2+D4=D3 OR D1+D3+D4=D2 OR D2+D3+D4=D1) OR
    { -- 乘法组合扩展 -- }
    (D1*D2 = D3*D4 AND PL>=1)
);

{ C. 三位数带两位小数运算 (关注 D0, D1, D2, D3, D4，百元股特有规律) }
MATH_3BIT := PL>=100 AND (
    { -- 五数求和为任意整十数 -- }
    (MOD(D0+D1+D2+D3+D4, 10)=0 AND (D0+D1+D2+D3+D4)>0)
);

{ D. 整数部分与小数位运算 (关注 INTPART, D3, D4，如 13.94: 13=9+4) }
INT_MATH := INTPART(PL) = (D3+D4) OR INTPART(PL) = (D3*D4);

{ 4. 汇总判定 }
SQ_ALL := BASE_4BIT OR BASE_3BIT OR MATH_1BIT OR MATH_2BIT OR MATH_3BIT OR INT_MATH;

{创60日新}
新低 := L = LLV(L, 10);  {条件1：20日新低}
{ 最终选股输出 }
七天试盘:= EXIST(C/REF(C,1) > 1.08, 8);
{当天是阳线}
阳线 := C > O;
{--- 综合输出 ---}
SELECT:阳线 AND 七天试盘 AND 新低 AND SQ_ALL AND S1_DONE AND IN_S2 AND BOX_SHAPE AND S2_DROP AND NEAR_BOTTOM AND S2_LEFT_BOTTOM;
